1 · Colar no Supabase

É a única coisa que só você pode fazer. Depois disto é um clique e acabou.

O que este comando faz
  • cria a tabela de tradução e as cinco colunas novas das funções
  • cria as duas tabelas de versão dos motores
  • cria a coluna dos quatro estados e as duas consultas de falta
Não apaga nada. Não altera nenhuma pontuação. Não troca nenhuma chave.
Pode rodar de novo sem medo — tudo é "se ainda não existir". A parte da tradução você já rodou às 01h06; ela vai passar batido.

Os 5 passos

  1. Clique no botão verde aqui embaixo.
  2. Abra o SQL Editor do Supabase.
  3. Clique dentro do editor e aperte Ctrl+A, depois Ctrl+V.
  4. Aperte Ctrl+Enter. Se pedir confirmação, confirme.
  5. Tem que aparecer Success. No rows returned.

Depois disso: volte para a pasta e dê dois cliques no 2-FAZER-A-TRANSFORMACAO.bat. Ele faz o resto sozinho.

O comando, se quiser ler

-- ============================================================
--  A TRANSFORMACAO  ·  16/08/2026  ·  ClubEfootball
--  So acrescenta. Nao apaga nada, nao muda pontuação, nao troca chave.
--  Pode rodar quantas vezes quiser.
-- ============================================================

-- ---------- 1) O TRADUTOR ----------
create table if not exists traducao (
  assunto        text        not null,
  chave          text        not null,
  fonte          text        not null,
  como_chama     text,
  medido         boolean     not null default false,
  observacao     text,
  atualizado_em  timestamptz not null default now(),
  primary key (assunto, chave, fonte)
);
create index if not exists traducao_assunto_idx on traducao (assunto);
create index if not exists traducao_como_idx    on traducao (como_chama);

-- ⛔ A coluna `nome` das funcoes CONTINUA sendo a chave. Nao mexemos nela:
--    tres tabelas apontam para ela com apagamento em cascata.
alter table funcoes add column if not exists codigo         text;
alter table funcoes add column if not exists rotulo         text;
alter table funcoes add column if not exists rotulo_curto   text;
alter table funcoes add column if not exists grupo          text;
alter table funcoes add column if not exists sigla_posicao  text;
create unique index if not exists funcoes_codigo_uk on funcoes (codigo) where codigo is not null;

-- ---------- 2) AS VERSOES DOS MOTORES ----------
create table if not exists motor_versao (
  motor          text        not null,
  versao         int         not null,
  quando         date,
  o_que_mudou    text,
  vigente        boolean     not null default false,
  atualizado_em  timestamptz not null default now(),
  primary key (motor, versao)
);

create table if not exists motor_arquivo (
  motor          text        not null,
  versao         int         not null,
  arquivo        text        not null,
  bytes          bigint,
  md5            text,
  modificado_em  timestamptz,
  atualizado_em  timestamptz not null default now(),
  primary key (motor, versao, arquivo)
);
create index if not exists motor_arquivo_md5_idx on motor_arquivo (md5);

create or replace view molde_confere as
select v.versao, v.nome, v.vigente,
       (select count(*) from molde m where m.versao = v.versao)                as linhas_no_banco,
       (select count(distinct m.funcao) from molde m where m.versao = v.versao) as funcoes_no_banco
from molde_versao v
order by v.versao;

-- ---------- 3) OS QUATRO ESTADOS ----------
alter table cards_base add column if not exists estado_de_cada_campo jsonb;
create index if not exists cards_base_estado_gin on cards_base using gin (estado_de_cada_campo);

create or replace view falta_de_verdade as
select e.key as campo, count(*) as cards
from cards_base c, lateral jsonb_each_text(c.estado_de_cada_campo) e
where e.value = 'nao_sei'
group by e.key
order by cards desc;

create or replace view o_que_puxar as
select c.card_id, c.nome, c.ovr, e.key as campo
from cards_base c, lateral jsonb_each_text(c.estado_de_cada_campo) e
where e.value = 'nao_sei';
